## 一、升级版系统目标与范围

### 1.1 核心目标

在基础版 MVP 的基础上，将系统从"算法选择器"升级为**面向路径规划领域的 AI Scientist 科研决策系统**。升级版实现以下核心能力：

- 不仅选算法，还能**推荐算法超参数**
- 从单一障碍物占比扩展为**多维度环境特征向量**
- 支持**动态场景下的自适应重规划**
- 用**机器学习模型**替代查表式评分
- 引入**10 指标多目标优化**评价函数
- 实现**"假设生成—实验验证—迭代优化"**闭环
- 支持**三维地图、拓扑图、连续空间**多地图类型
- 支持**多机器人协同**路径规划
- 输出**完整推理链**（环境→数据库→排除→选择→风险）

```python
# agents/parameter_agent.py — 参数推荐智能体

import json


class ParameterAgent:
    """根据场景等级和选定算法，推荐最优超参数"""

    def __init__(self, templates_path: str = "knowledge_base/parameter_templates.json"):
        with open(templates_path, "r", encoding="utf-8") as f:
            self.templates = json.load(f)

    def recommend(self, scene_level: str, algorithm: str) -> dict:
        algo_params = self.templates.get(algorithm, {})
        return algo_params.get(scene_level, {})

    def inject(self, algorithm_instance, scene_level: str, algorithm: str):
        """动态注入参数到算法实例"""
        params = self.recommend(scene_level, algorithm)
        for key, value in params.items():
            if hasattr(algorithm_instance, key):
                setattr(algorithm_instance, key, value)
        return algorithm_instance
```

## 四、优化二：多维度环境特征建模

### 4.1 目标

从单一 `obstacle_ratio` 扩展为复合特征向量，包含障碍聚集度、离散度、通道宽度、死路数量、连通区域数等。

### 4.2 创建 `map/map_analyzer.py`

```python
# map/map_analyzer.py — 地图结构分析器

import numpy as np
from collections import deque


class MapAnalyzer:
    """从栅格地图中提取多维度环境特征"""

    DIRECTIONS = [(-1, 0), (1, 0), (0, -1), (0, 1)]

    def analyze(self, grid: np.ndarray, start: tuple, target: tuple) -> dict:
        h, w = grid.shape
        features = {}

        features["obstacle_ratio"] = round(grid.sum() / grid.size, 4)

        # 障碍聚集度
        cluster_count = 0
        for y in range(h):
            for x in range(w):
                if grid[y, x] == 1:
                    for dx, dy in [(1, 0), (0, 1)]:
                        nx, ny = x + dx, y + dy
                        if 0 <= nx < w and 0 <= ny < h and grid[ny, nx] == 1:
                            cluster_count += 1
        max_pairs = (h * (w - 1) + w * (h - 1))
        features["obstacle_clustering"] = round(cluster_count / max(max_pairs, 1), 4)

        # 障碍离散度
        obstacle_positions = np.argwhere(grid == 1)
        if len(obstacle_positions) > 0:
            features["obstacle_dispersion"] = round(
                np.std(obstacle_positions[:, 0]) / h +
                np.std(obstacle_positions[:, 1]) / w, 4
            )
        else:
            features["obstacle_dispersion"] = 0.0

        # 连通区域计数
        features["connected_regions"] = self._count_regions(grid)

        # 死路数量
        dead_ends = 0
        for y in range(h):
            for x in range(w):
                if grid[y, x] == 0:
                    neighbors = sum(
                        1 for dx, dy in self.DIRECTIONS
                        if 0 <= x + dx < w and 0 <= y + dy < h
                        and grid[y + dy, x + dx] == 0
                    )
                    if neighbors == 1:
                        dead_ends += 1
        features["dead_ends"] = dead_ends

        # 平均值通道宽度
        features["avg_passage_width"] = round(self._estimate_passage_width(grid), 2)

        # 起点终点直线距离
        straight_dist = np.sqrt(
            (target[0] - start[0]) ** 2 + (target[1] - start[1]) ** 2
        )
        features["straight_distance"] = round(straight_dist, 1)

        return features

    def _count_regions(self, grid: np.ndarray) -> int:
        h, w = grid.shape
        visited = np.zeros_like(grid, dtype=bool)
        regions = 0
        for y in range(h):
            for x in range(w):
                if grid[y, x] == 0 and not visited[y, x]:
                    regions += 1
                    queue = deque([(x, y)])
                    visited[y, x] = True
                    while queue:
                        cx, cy = queue.popleft()
                        for dx, dy in self.DIRECTIONS:
                            nx, ny = cx + dx, cy + dy
                            if (0 <= nx < w and 0 <= ny < h
                                    and not visited[ny, nx]
                                    and grid[ny, nx] == 0):
                                visited[ny, nx] = True
                                queue.append((nx, ny))
        return regions

    def _estimate_passage_width(self, grid: np.ndarray) -> float:
        h, w = grid.shape
        widths = []
        for y in range(h):
            run = 0
            for x in range(w):
                if grid[y, x] == 0:
                    run += 1
                else:
                    if run > 0:
                        widths.append(run)
                    run = 0
            if run > 0:
                widths.append(run)
        return float(np.mean(widths)) if widths else 0
```

## 五、优化三：动态场景重规划机制

### 5.1 三级响应策略

| 变化率 | 响应策略 | 耗时 | 适用场景 |
|--------|----------|------|----------|
| < 10% | **局部路径修正**：仅局部调整路径 | 低 | 少量新障碍 |
| 10%~30% | **保持算法重新搜索**：不改变算法，重新运行 | 中 | 通道部分阻塞 |
| > 30% | **重新全流程选型**：重新评估+选算法+选参数 | 高 | 地图结构大改 |

| 创建重规划决策智能体 | `agents/replanning_agent.py` | 三级响应策略判断 |
| 创建局部路径修正器 | `algorithms/local_repair.py` | A* 局部绕行修复 |

#### `map/dynamic_obstacle.py`

```python
import numpy as np

class DynamicObstacleSimulator:
    def __init__(self, seed: int = None):
        self.rng = np.random.RandomState(seed)
    def apply_change(self, grid: np.ndarray, change_rate: float) -> np.ndarray:
        new_grid = grid.copy()
        h, w = grid.shape
        passable = np.argwhere(grid == 0)
        n_changes = int(len(passable) * change_rate)
        chosen = self.rng.choice(len(passable), size=n_changes, replace=False)
        for idx in chosen:
            y, x = passable[idx]
            new_grid[y, x] = 1
        return new_grid
```

```python
class ReplanningAgent:
    LOW_THRESHOLD = 0.10
    MEDIUM_THRESHOLD = 0.30
    def decide(self, change_ratio: float, current_algorithm: str, has_path: bool) -> dict:
```

#### `algorithms/local_repair.py`

```python
class LocalPathRepair:
    def repair(self, grid, original_path, start, target):
        collision_indices = [
            i for i, (x, y) in enumerate(original_path)
            if not (0 <= x < grid.shape[1] and 0 <= y < grid.shape[0])
            or grid[y, x] == 1
        ]
        if not collision_indices:
            return original_path
        # 对每个碰撞段调用 A* 局部绕行
        from algorithms.astar import AStar
        astar = AStar()
        repaired = list(original_path)
        for idx in collision_indices:
            before = max(0, idx - 5)
            after = min(len(repaired) - 1, idx + 5)
            result = astar.search(grid, repaired[before], repaired[after])
            if result["success"]:
                repaired = repaired[:before] + result["path"] + repaired[after + 1:]
        return repaired
```

```python
from abc import ABC, abstractmethod
import numpy as np

class BasePredictor(ABC):
    @abstractmethod
    def train(self, X: np.ndarray, y: np.ndarray): pass
    @abstractmethod
    def predict(self, features: dict) -> dict: pass

class RuleBasedPredictor(BasePredictor):
    def __init__(self, knowledge_agent):
        self.knowledge_agent = knowledge_agent
    def train(self, X, y): pass
    def predict(self, features: dict, scene_level: str) -> dict:
        records = self.knowledge_agent.retrieve(scene_level)
        return {r["algorithm"]: r for r in records}
```

#### `evaluation/models/random_forest.py`

```python
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from evaluation.predictor import BasePredictor

class RandomForestPredictor(BasePredictor):
    def __init__(self):
        self.time_model = RandomForestRegressor(n_estimators=100)
        self.success_model = RandomForestRegressor(n_estimators=100)
        self.length_model = RandomForestRegressor(n_estimators=100)

    def train(self, X, y):
        self.time_model.fit(X, y[:, 0])
        self.success_model.fit(X, y[:, 1])
        self.length_model.fit(X, y[:, 2])

    def predict(self, features: np.ndarray) -> dict:
        f = features.reshape(1, -1)
        return {
            "predicted_time": round(float(self.time_model.predict(f)[0]), 4),
```

```python
class ReflectionAgent:
    """检查决策结论的可信度与自洽性"""

    def check(self, decision: dict, experiment_results: list[dict]) -> dict:
        issues = []
        recommended = decision["best_algorithm"]
        if experiment_results:
            best_actual = max(experiment_results, key=lambda r: r.get("success_rate", 0))
            if best_actual["algorithm"] != recommended:
                issues.append(
                    f"系统推荐 {recommended}，但实验显示 {best_actual['algorithm']} 成功率更高。"
                )

        ranking = decision.get("ranking", [])
        if len(ranking) >= 2 and ranking[0]["score"] - ranking[1]["score"] < 0.03:
            issues.append("前两名算法评分过于接近，建议输出备选方案。")

        return {
            "is_reliable": len(issues) == 0,
            "issues": issues,
            "suggestion": "当前决策可信" if not issues else "\n".join(issues),
        }
```

```python
import numpy as np

class MultiAgentCoordinator:
    def plan(self, grid, tasks, base_algorithm) -> list:
        tasks_sorted = sorted(tasks, key=lambda t: t.get("priority", 0))
        reservations = {}
        plans = []
        for task in tasks_sorted:
            result = base_algorithm.search(grid, task["start"], task["target"])
            if result["success"]:
                adjusted = self._resolve_conflicts(
                    result["path"], reservations, task.get("id", 0)
                )
                plans.append({"robot_id": task["id"], "path": adjusted})
            else:
                plans.append({"robot_id": task["id"], "error": "无可行路径"})
        return plans

    def _resolve_conflicts(self, path, reservations, robot_id):
        resolved = []
        t = 0
        for point in path:
            while point in reservations.get(t, {}):
                resolved.append(resolved[-1] if resolved else point)
                t += 1
            reservations.setdefault(t, {})[point] = robot_id
            resolved.append(point)
            t += 1
        return resolved
```

```python
def explain_full_chain(self, env: dict, decision: dict) -> dict:
    ranking = decision["ranking"]
    best = ranking[0]
    chain = {
        "1_环境分析": f"障碍物占比{env.get('obstacle_ratio',0):.0%} → {decision['scene_level']}",
        "2_数据库检索": f"找到{len(ranking)}条{decision['scene_level']}记录",
        "3_规则排除": self._format_exclusions(ranking),
        "4_算法对比": " | ".join(f"{r['algorithm']}得分{r['score']:.2f}" for r in ranking),
        "5_最优选择": f"{best['algorithm']}得分{best['score']}，成功率{best.get('success_rate','N/A')}",
        "6_风险评估": self._format_risk(ranking),
    }
    return chain

def _format_exclusions(self, ranking):
    parts = [f"{r['algorithm']}惩罚{'-' + str(r['penalty'])}" for r in ranking if r.get('penalty', 0) > 0]
    return " | ".join(parts) if parts else "无惩罚"

def _format_risk(self, ranking):
    gap = ranking[0]["score"] - ranking[1]["score"] if len(ranking) >= 2 else 1
    if gap < 0.05:
        return f"前两名差距{gap:.3f}<0.05，建议人工校验"
    return f"前两名差距{gap:.3f}>0.05，无需人工校验"
```

```python
def draw_reasoning_flow(chain: dict) -> go.Figure:
    stages = list(chain.keys())
    descriptions = list(chain.values())
    colors = ["#3498db", "#2ecc71", "#e74c3c", "#f39c12", "#9b59b6", "#1abc9c"]

        )

                               text="↓", showarrow=False, font=dict(size=16))
```
